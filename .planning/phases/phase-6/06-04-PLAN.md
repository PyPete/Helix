      export function exportToWsl(editor: NodeEditor<ClassicPreset.Schemes>, name: string): WslDocument {
        const nodes: WslNode[] = editor.getNodes().map(n => ({
          nodeId: n.id,
          label: n.label,
          nodeType: (n as HelixNode).nodeType ?? "Step",
          parameters: (n as HelixNode).getParameters ? (n as HelixNode).getParameters() : {},
        }));
        const edges: WslEdge[] = editor.getConnections().map(c => ({
          edgeId: c.id,
          sourceNodeId: c.source,
          targetNodeId: c.target,
        }));
        return { documentId: crypto.randomUUID(), name, schemaVersion: "1.0", nodes, edges };
      }
