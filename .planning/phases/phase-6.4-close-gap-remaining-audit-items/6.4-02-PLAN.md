  interface CampaignsPageProps { baseUrl: string; }
  export function CampaignsPage({ baseUrl }: CampaignsPageProps) {
    const [campaigns, setCampaigns] = useState<CampaignDto[]>([]);
    const [loading, setLoading] = useState(false);
    const [error, setError] = useState<string | null>(null);
    useEffect(() => { void loadCampaigns(); }, [baseUrl]);
    const loadCampaigns = async () => {
      setLoading(true); setError(null);
      try {
        const res = await fetch(baseUrl + "/api/campaigns");
        if (!res.ok) { setError("Failed to load campaigns: " + res.statusText); return; }
        setCampaigns(await res.json() as CampaignDto[]);
      } catch (e) { setError("Network error"); }
      finally { setLoading(false); }
    };
  }
