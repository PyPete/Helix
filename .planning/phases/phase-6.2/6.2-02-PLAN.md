    Create helix-desktop/src/hooks/useNodeExecutionState.ts:

    ```typescript
    import { useMemo } from "react";
    import type { RunspaceEvent } from "../types/agent";
    import type { ExecutionState } from "../components/WorkflowBuilder/nodes/HelixNodes";
    import type { Edge } from "reactflow";

    export interface NodeExecutionState {
      executionState: ExecutionState;
      startedAt?: Date;
      estimatedCompletionAt?: Date;
      finishedAt?: Date;
      faultMessage?: string;
    }

    export interface UseNodeExecutionStateResult {
      nodeStates: Record<string, NodeExecutionState>;
      edgeExecutionStates: (edges: Edge[]) => Record<string, ExecutionState>;
    }

    export function useNodeExecutionState(
      runspaceEvents: RunspaceEvent[]
    ): UseNodeExecutionStateResult {
      const nodeStates = useMemo(() => {
        const states: Record<string, NodeExecutionState> = {};
        for (const event of runspaceEvents) {
          const stepId = String(event["stepId"] ?? "");
          if (event.type === "runspace-started") {
            // Clear all states
            for (const key of Object.keys(states)) delete states[key];
          } else if (event.type === "step-started" && stepId) {
            const estimatedMs = event.estimatedDurationMs;
            states[stepId] = {
              executionState: "running",
              startedAt: new Date(event.timestamp),
              estimatedCompletionAt: estimatedMs != null
                ? new Date(Date.parse(event.timestamp) + estimatedMs)
                : undefined,
            };
          } else if (event.type === "step-completed" && stepId) {
            states[stepId] = {
              executionState: "succeeded",
              finishedAt: new Date(event.timestamp),
            };
          } else if (event.type === "step-faulted" && stepId) {
            states[stepId] = {
              executionState: "failed",
              faultMessage: String(event["faultMessage"] ?? ""),
            };
          }
        }
        return states;
      }, [runspaceEvents]);

      const edgeExecutionStates = useMemo(
        () =>
          (edges: Edge[]): Record<string, ExecutionState> => {
            const result: Record<string, ExecutionState> = {};
            for (const edge of edges) {
              result[edge.id] = nodeStates[edge.source]?.executionState ?? "idle";
            }
            return result;
          },
        [nodeStates]
      );

      return { nodeStates, edgeExecutionStates };
    }
    ```

    ```typescript
    function formatHHMMSS(ms: number): string {
      const totalSec = Math.max(0, Math.floor(ms / 1000));
      const h = Math.floor(totalSec / 3600);
      const m = Math.floor((totalSec % 3600) / 60);
      const s = totalSec % 60;
      return [h, m, s].map((v) => String(v).padStart(2, "0")).join(":");
    }

    interface NodeTimingFooterProps {
      executionState?: ExecutionState;
      startedAt?: string;
      estimatedCompletionAt?: string;
      finishedAt?: string;
      faultMessage?: string;
    }

    const footerStyle: React.CSSProperties = {
      padding: "4px 10px",
      fontSize: 10,
      borderTop: "1px solid #2a2a3e",
      color: "#6c7086",
      fontFamily: "monospace",
    };

    function NodeTimingFooter({ executionState, startedAt, estimatedCompletionAt, finishedAt }: NodeTimingFooterProps) {
      if (!executionState || executionState === "idle") return null;

      const now = Date.now();

      if (executionState === "running") {
        const elapsed = startedAt ? now - Date.parse(startedAt) : 0;
        const etaPart = estimatedCompletionAt
          ? `  ETA ~${formatHHMMSS(Date.parse(estimatedCompletionAt) - now)}`
          : "";
        return (
          <div style={{ ...footerStyle, color: "#f5a623" }} data-testid="timing-footer-running">
            {`â–¶ ${formatHHMMSS(elapsed)}${etaPart}`}
          </div>
        );
      }

      if (executionState === "succeeded") {
        const durationMs = (startedAt && finishedAt)
          ? Date.parse(finishedAt) - Date.parse(startedAt)
          : 0;
        const totalSec = Math.max(0, Math.floor(durationMs / 1000));
        const mins = Math.floor(totalSec / 60);
        const secs = totalSec % 60;
        const label = mins > 0 ? `${mins}m ${secs}s` : `${secs}s`;
        return (
          <div style={{ ...footerStyle, color: "#27ae60" }} data-testid="timing-footer-succeeded">
            {`âœ“ Finished in ${label}`}
          </div>
        );
      }

      if (executionState === "failed") {
        const timeStr = finishedAt
          ? new Date(finishedAt).toTimeString().slice(0, 8)
          : "--:--:--";
        return (
          <div style={{ ...footerStyle, color: "#e74c3c" }} data-testid="timing-footer-failed">
            {`âœ— Failed at ${timeStr}`}
          </div>
        );
      }

      return null;
    }
    ```
