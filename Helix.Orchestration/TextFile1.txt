🎯 Phase 2 — Agent Runtime with Language-Agnostic Adapter Discovery
Context & Architectural Decisions
Phase 1 delivered the core domain contracts (trends, leasing, WSL, runspace, TestContext) with 56 passing tests.

Phase 2 builds the Agent runtime that executes workflows with production-grade durability and discovers language-agnostic device adapters managed by the operating system.

Key Architectural Decisions (User-Confirmed)
Adapter IPC: gRPC with .proto contracts
Adapter Lifecycle: OS-managed (systemd/Windows Service Manager); Agent discovers via mDNS/registry/config
Device Registry: Local JSON file with Site Server sync (offline-first for safety)
Trend Routing: Agent as MQTT gateway (adapters → gRPC → agent → MQTT)
Cleipnir Integration: Wrap Cleipnir Workflow inside IRunspace abstraction
Phase 2 Adapter Scope: Build one real Python adapter (simulated pump); defer Node.js/C++ to later phases
Why This Architecture Matters
This is the industrial IoT model used by:

OPC UA servers (always-on system services)
SiLA 2 devices (independent HTTP/gRPC endpoints)
PLC gateways (dedicated controllers)
Robotics controllers (ROS nodes)
Instrument control daemons (vendor-supplied services)
Benefits:

Adapters auto-start on boot, auto-restart on crash
Agent stays stateless and simple
Adapters can run on same machine, USB Bridge Agent, remote device, or container
Crash isolation: adapter crash ≠ agent crash
Language freedom: Python, Node.js, C++, Rust all equal

public class AdapterProxy
{
    private readonly GrpcChannel _channel;
    private readonly DeviceAdapter.DeviceAdapterClient _client;
    
    public async Task<CapabilityResponse> InvokeAsync(
        string capabilityName, 
        object? parameters, 
        string leaseId,
        CancellationToken ct)
    {
        var request = new CapabilityRequest
        {
            CapabilityName = capabilityName,
            ParametersJson = JsonSerializer.Serialize(parameters),
            LeaseId = leaseId
        };
        return await _client.InvokeCapabilityAsync(request, cancellationToken: ct);
    }
    
    public IAsyncEnumerable<TrendValue> StreamTrendsAsync(CancellationToken ct)
    {
        var call = _client.StreamTrends(new Empty(), cancellationToken: ct);
        return call.ResponseStream.ReadAllAsync(ct);
    }
}
DurableRunspace.cs — Production runspace:

public class DurableRunspace : IRunspace
{
    private readonly FunctionsRegistry _registry;
    private readonly IExperimentContext _experimentContext;
    private readonly IDeviceLeasingService _leasingService;
    private readonly ICapabilityInvoker _capabilities;
    private Workflow? _workflow;
    
    public async Task StartAsync(RunspaceDefinition definition, CancellationToken ct)
    {
        // Compile definition into executable delegate
        var workflowDelegate = CompileDefinition(definition);
        
        // Register with Cleipnir
        var registration = _registry.RegisterAction<string>(definition.Name);
        
        // Invoke (creates durable instance)
        _workflow = await registration.Invoke(definition.DefinitionId, workflowDelegate);
    }
    
    private Func<string, Task> CompileDefinition(RunspaceDefinition definition)
    {
        // Transform WSL nodes/edges into executable workflow
        return async (instanceId) =>
        {
            foreach (var node in definition.Nodes)
            {
                // Execute node via CapabilityInvoker
                await _capabilities.InvokeAsync(node.NodeType, node.Parameters);
                
                // Record to ExperimentContext
                await _experimentContext.RecordDataAsync(node.NodeId, node.Parameters);
            }
        };
    }
}
MqttTrendPublisher.cs — Agent → MQTT gateway:

public class MqttTrendPublisher : IHostedService
{
    private readonly IMqttClient _client;
    private readonly TrendAggregator _aggregator;
    
    public async Task StartAsync(CancellationToken ct)
    {
        // Connect to MQTT broker
        await _client.ConnectAsync(new MqttClientOptionsBuilder()
            .WithTcpServer(_config.BrokerHost, _config.BrokerPort)
            .WithTls()
            .Build(), ct);
        
        // Subscribe to adapter trend streams
        await _aggregator.StartAsync(ct);
    }
    
    public async Task PublishAsync(string experimentId, TrendValue value, CancellationToken ct)
    {
        var topic = $"helix/{_config.SiteId}/{_config.FumehoodId}/{experimentId}/trends/{value.TrendId}";
        var payload = JsonSerializer.SerializeToUtf8Bytes(value);
        
        await _client.PublishAsync(new MqttApplicationMessageBuilder()
            .WithTopic(topic)
            .WithPayload(payload)
            .WithQualityOfServiceLevel(MqttQualityOfServiceLevel.AtLeastOnce)
            .Build(), ct);
    }
}
3. Python Adapter (Real, OS-Managed)

Implementation:

#!/usr/bin/env python3
import grpc
from concurrent import futures
import json
import time
import logging
from generated import adapter_pb2, adapter_pb2_grpc

class PumpSimulatorAdapter(adapter_pb2_grpc.DeviceAdapterServicer):
    def __init__(self):
        self.flow_rate = 0.0
        self.is_running = False
        self.capabilities = ["SetFlowRate", "GetFlowRate", "Start", "Stop"]
    
    def GetCapabilities(self, request, context):
        return adapter_pb2.CapabilitiesResponse(capabilities=self.capabilities)
    
    def InvokeCapability(self, request, context):
        cap_name = request.capability_name
        params = json.loads(request.parameters_json) if request.parameters_json else {}
        
        if cap_name == "SetFlowRate":
            self.flow_rate = params.get("flowRate", 0.0)
            return adapter_pb2.CapabilityResponse(
                success=True,
                result_json=json.dumps({"flowRate": self.flow_rate})
            )
        elif cap_name == "Start":
            self.is_running = True
            return adapter_pb2.CapabilityResponse(success=True)
        elif cap_name == "Stop":
            self.is_running = False
            return adapter_pb2.CapabilityResponse(success=True)
        elif cap_name == "GetFlowRate":
            return adapter_pb2.CapabilityResponse(
                success=True,
                result_json=json.dumps({"flowRate": self.flow_rate, "running": self.is_running})
            )
        else:
            return adapter_pb2.CapabilityResponse(
                success=False,
                error_message=f"Unknown capability: {cap_name}"
            )
    
    def StreamTrends(self, request, context):
        """Simulate flow rate trend."""
        while True:
            if self.is_running:
                yield adapter_pb2.TrendValue(
                    trend_id="pump-01-flow",
                    trend_kind="Scalar",
                    value_json=json.dumps({"value": self.flow_rate, "units": "mL/min"}),
                    timestamp=int(time.time() * 1000)
                )
            time.sleep(1.0)
    
    def HealthCheck(self, request, context):
        return adapter_pb2.HealthStatus(healthy=True, message="Pump simulator online")

def serve():
    server = grpc.server(futures.ThreadPoolExecutor(max_workers=10))
    adapter_pb2_grpc.add_DeviceAdapterServicer_to_server(PumpSimulatorAdapter(), server)
    server.add_insecure_port('[::]:50051')
    logging.info("Pump Simulator Adapter listening on port 50051")
    server.start()
    server.wait_for_termination()

if __name__ == '__main__':
    logging.basicConfig(level=logging.INFO)
    serve()

    [Fact]

    {
        // Arrange
        var wslDoc = new WslDocument
        {
            Name = "Pump Test",

        // Act

    }

    {
        await _agentService?.StopAsync(CancellationToken.None)!;
        _pythonAdapterProcess?.Kill();
        _pythonAdapterProcess?.Dispose();
    }

services.AddSingleton<IFunctionStore>(sp =>
{
    var config = sp.GetRequiredService<AgentConfiguration>();
    return config.Cleipnir.StoreType switch
    {
        "SQLite" => new SqliteFunctionStore(config.Cleipnir.ConnectionString),
        "PostgreSQL" => new PostgreSqlFunctionStore(config.Cleipnir.ConnectionString),
        _ => throw new InvalidOperationException($"Unknown store type: {config.Cleipnir.StoreType}")
    };
});
